FUNÇÃO SUBSTRING - PARA MOSTRAR UMA LETRA (DETERMINANDO A POSIÇÃO INICIAL E O TAMANHO DE LETRAS QUE VOU USAR APÓS O INICIAL)

SELECT 'SR.(a)' || SUBSTRING(first_name, 1, 1) || '. ' || last_name AS Função

FROM Customers

;


FUNÇÃO LENGTH - PARA BUSCAR A QUANTIDADE DE LETRAS DE UMA COLUNA DE NOMES POR EXEMPLO


SELECT first_name, LENGTH(first_name) AS n_letras

FROM Customers;


-> BUSCANDO O NOME QUE TEM MAIS LETRAS NA TABELA

SELECT first_name

FROM Customers

WHERE LENGTH(first_name) > 5

;


-> UM VERDADEIRO E FALSO QUE MOSTRA A COLUNA DE NOME E CRIA UMA COLUNA AO LADO EM BOOLEANO MOSTRANDO COM 0 PARA AQUELES QUE NÃO TEM MAIS QUE 5 LETRAS E 1 PARA AS QUE TEM MAIS DE 5 LETRAS.

SELECT first_name, LENGTH(first_name) > 5 AS tem_mais_de_5_letras 

FROM Customers


;


--FUNÇÕES DE AGREGAÇÃO - MAX EM IDADE

SELECT first_name, max(age) AS maior_idade

FROM Customers;


--FUNÇÕES DE AGREGAÇÃO - MAX EM NOMES 

SELECT first_name, max(LENGTH(first_name)) AS MAIOR

FROM Customers;


--FUNÇÕES DE AGREGAÇÃO - AVG FAZ O CÁLCULO DA MÉDIA DE ALGO NA TABELA

SELECT AVG(age) AS média_idades

FROM Customers;


--FUNÇÕES DE AGREGAÇÃO - COUNT ELE CONTA A QUANTIDADE DE ALGO

SELECT COUNT(first_name) AS n_linhas

FROM Customers;


--FUNÇÃO DE AGREGAÇÃO - SUM ELE FAZ A SOMA. NESSE CASO ESTOU FAZENDO A SOMA DE TODAS A IDADE DA TABELA CUSTOMERS

SELECT SUM(age) AS soma_das_idade

FROM Customers;


--FUNÇÕES DE AGREGAÇÃO - NESSA BUSCA ELE VAI DA A QUANTIDADE DE PESSOAS DE ACORDO COM O PRIMEIRO NOME

SELECT COUNT(first_name) AS n_linhas, first_name

FROM Customers

GROUP BY first_name; --AGRUPAR POR ALGUMA COLUNA (ELE AGRUPA VALORES IGUAIS)


--FUNÇÕES DE AGREGAÇÃO - NESSA BUSCA ELE VAI DA A MÉDIA DA IDADE DE CADA PESSOA DE ACORDO COM O PRIMEIRO NOME

SELECT AVG(age) AS media_idade, first_name

FROM Customers

GROUP BY first_name; --AGRUPAR POR ALGUMA COLUNA (ELE AGRUPA VALORES IGUAIS)


-- EXERCÍCIO: CALCULE A MÉDIA DA IDADE DE CADA CLIENTE POR PAÍS

SELECT AVG(age) AS média_idade, country

FROM Customers

GROUP BY country;


-- EXERCÍCIO: QUANTOS CLIENTE TEM CADA IDADE?

SELECT COUNT(age) AS Clientes, age 

FROM Customers

GROUP BY age;


-- FUNÇÕES DE AGREGAÇÃO:ORDER BY / DESC / ASC

SELECT COUNT(age) AS Clientes, age 

FROM Customers

GROUP BY age

ORDER BY Clientes DESC; --UTILIZANDO ESSA FUNÇÃO VOCÊ ORDENAR A SAÍDA SEJA ELA EM ORDEM DO MAIOR AO MENOR OU AO CONTRÁRIO OU EM ORDEM ALFABÉTICA DEPENDE DO CASO - 'DESC' (DESCENDENTE) -> DO MAIOR PARA O MENOR E 'ASC' (ASCEDENTE) DO MENOR PARA O MAIOR



-- EXERCÍCIO: 01 - MÉDIA DAS IDADE DOS CLIENTES POR ITEM VENDIDOS

SELECT item, AVG(age) AS médias

FROM Orders

JOIN Customers

ON Orders.customer_id = Customers.customer_id

GROUP BY item;


-- EXERCÍCIO: 02 - QUANTIDADE DE ENTREGAS PENDENTES
